# 公信万年历 — 前后端分离实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** 将单文件 HTML（17127行）拆分为 core 共享计算库 + Node.js API 后端 + 原生 JS 前端三层架构

**Architecture:** Monorepo 单仓库，core/ 作为 npm 本地包被 server/ 和 web/ 共享引用。server/ 提供 REST API，web/ 通过 api.js 客户端调用。模块间通过 state.js 事件总线通信，UI 模块零直接依赖。

**Tech Stack:** Node.js v24 + Express + lunisolar@2.6.0 + @lunisolar/plugin-thegods@0.1.4 + Vite + 原生 JS (ES5)

```javascript
// server/middleware/auth.js
const crypto = require('crypto');

const HMAC_SECRET = process.env.HMAC_SECRET || '98964eb12554b259863a7ebc40a82562';
const TIME_WINDOW = 5 * 60 * 1000; // ±5分钟
const NONCE_CACHE = new Map(); // LRU 用 Map 模拟
const NONCE_MAX = 10000;

async function hmacAuth(req, res, next) {
  var mid = req.headers['x-machine-id'];
  var ts = parseInt(req.headers['x-timestamp'], 10);
  var nonce = req.headers['x-nonce'];
  var sig = req.headers['x-signature'];

  if (!mid || !ts || !nonce || !sig) {
    return res.status(401).json({ ok: false, error: 'Missing auth headers', code: 401 });
  }

  // 时间窗口检查
  if (Math.abs(Date.now() - ts) > TIME_WINDOW) {
    return res.status(401).json({ ok: false, error: 'Timestamp expired', code: 401 });
  }

  // nonce 去重
  if (NONCE_CACHE.has(nonce)) {
    return res.status(401).json({ ok: false, error: 'Nonce reused', code: 401 });
  }

  // 签名验证
  var message = mid + ts + nonce + req.path;
  var key = await crypto.subtle.importKey('raw',
    new TextEncoder().encode(HMAC_SECRET),
    { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  var expectedBuf = await crypto.subtle.sign('HMAC', key,
    new TextEncoder().encode(message));
  var expectedHex = Array.from(new Uint8Array(expectedBuf.slice(0, 4)))
    .map(b => b.toString(16).padStart(2, '0')).join('');

  if (expectedHex !== sig.toLowerCase()) {
    return res.status(401).json({ ok: false, error: 'Invalid signature', code: 401 });
  }

  // 记录 nonce
  NONCE_CACHE.set(nonce, Date.now());
  if (NONCE_CACHE.size > NONCE_MAX) {
    var oldest = NONCE_CACHE.keys().next().value;
    NONCE_CACHE.delete(oldest);
  }

  req.machineId = mid;
  next();
}

module.exports = hmacAuth;
```

```javascript
// server/middleware/ratelimit.js
module.exports = function createRateLimiter(opts) {
  var windowMs = (opts && opts.windowMs) || 60000;
  var max = (opts && opts.max) || 60;
  var hits = {};

  // 定期清理
  setInterval(function() {
    var now = Date.now();
    for (var ip in hits) {
      if (now - hits[ip].reset > windowMs * 2) delete hits[ip];
    }
  }, windowMs).unref();

  return function rateLimit(req, res, next) {
    var ip = req.ip || req.connection.remoteAddress;
    var now = Date.now();
    var entry = hits[ip];

    if (!entry || now - entry.reset > windowMs) {
      hits[ip] = { count: 1, reset: now };
      return next();
    }

    entry.count++;
    if (entry.count > max) {
      return res.status(429).json({
        ok: false, error: 'Rate limit exceeded. Try again in ' +
          Math.ceil((entry.reset + windowMs - now) / 1000) + 's', code: 429
      });
    }
    next();
  };
};
```

```javascript
// web/js/state.js — 发布/订阅事件总线
const State = (function() {
  var _listeners = {};
  var _cache = {};

  function on(event, fn) {
    if (!_listeners[event]) _listeners[event] = [];
    _listeners[event].push(fn);
  }

  function off(event, fn) {
    var arr = _listeners[event];
    if (arr) {
      _listeners[event] = arr.filter(function(f) { return f !== fn; });
    }
  }

  function emit(event, data) {
    _cache[event] = data;
    var arr = _listeners[event];
    if (arr) arr.forEach(function(fn) {
      try { fn(data); } catch(e) { console.error('[state] error in listener for ' + event, e); }
    });
  }

  function getCache(event) { return _cache[event]; }

  return { on: on, off: off, emit: emit, getCache: getCache, cache: _cache };
})();
```

```javascript
    var keyBuf = new TextEncoder().encode(HMAC_KEY);
    var key = await crypto.subtle.importKey('raw', keyBuf,
      { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
    var sigBuf = await crypto.subtle.sign('HMAC', key,
      new TextEncoder().encode(message));
    return Array.from(new Uint8Array(sigBuf.slice(0, 4)))
      .map(function(b) { return b.toString(16).padStart(2, '0'); }).join('');
  }

  async function post(path, data) {
    var ts = Date.now();
    var nonce = Array.from(crypto.getRandomValues(new Uint8Array(4)))
      .map(function(b) { return b.toString(16).padStart(2, '0'); }).join('');
    var mid = localStorage._license_mid || 'web-unknown';
    var sig = await _sign(mid, ts, nonce, path);

    var resp = await fetch(BASE + path, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Machine-Id': mid,
        'X-Timestamp': ts,
        'X-Nonce': nonce,
        'X-Signature': sig
      },
      body: JSON.stringify(data)
    });

    if (!resp.ok) {
      var err = await resp.json().catch(function() { return { error: 'Network error' }; });
      throw new Error(err.error || 'HTTP ' + resp.status);
    }
    return resp.json();
  }

  return { post: post };
})();
```

```javascript
        return;
      }
    }
```

```javascript
// web/sw.js — Service Worker
var CACHE_NAME = 'gongxin-v1';
var PRE_CACHE = [
  '/', '/index.html',
  '/css/variables.css', '/css/base.css', '/css/layout.css', '/css/calendar.css',
  '/js/app.js', '/js/state.js', '/js/api.js', '/js/router.js', '/js/dom-helpers.js', '/js/calendar-ui.js',
  '/assets/icons/icon-192.png', '/assets/icons/icon-512.png'
];

self.addEventListener('install', function(event) {
  event.waitUntil(
    caches.open(CACHE_NAME).then(function(cache) { return cache.addAll(PRE_CACHE); })
  );
});

self.addEventListener('fetch', function(event) {
  // API请求走网络，静态资源走缓存
  if (event.request.url.includes('/api/')) {
    event.respondWith(
      fetch(event.request).catch(function() {
        return new Response(JSON.stringify({ ok: false, error: 'Offline', code: 0 }),
          { headers: { 'Content-Type': 'application/json' } });
      })
    );
  } else {
    event.respondWith(
      caches.match(event.request).then(function(cached) {
        return cached || fetch(event.request);
      })
    );
  }
});
```
